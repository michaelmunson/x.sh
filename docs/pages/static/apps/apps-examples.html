Comprehensive example app (exapp)

The app definition imports scripts from two sidecar files via import.$:

exapp.x.yml 
# Comprehensive sample app for the `x` app framework (`x exapp …`).
# Demonstrates: metadata, root options/arguments, nested commands, YAML anchors,
# every synopsis form from the cheat-sheet, and split `import.$` handler files.

name: exapp
version: 0.1.0
description: >-
  Exercise app - exercises options, positionals, nesting, validation, and bash helpers
  (`x-opt`, `x-arg`, `x-opts`, `x-args`, `x-run`, `x-usage`).
import:
  $:
    - handlers/exapp.handlers-a.yml
    - handlers/exapp.handlers-b.yml
env:
  DEBUG: "true"

options:
  - '[-v | --version]'
  - '[-g | --global <note="">]'

arguments: '[<topic="overview">]'

.demo:
  description: nested demos (each subcommand highlights different DSL pieces)

  .opts:
    description: >-
      Option forms - bool pair, value, defaulted value, repeating values,
      enum choice, `=` default, requires-chain, and a required bare flag.
    opts: |
      [-n | --dry-run]
      [-o | --out <path>]
      [-c | --count <n="1">]
      [-D | --define <kv> ...]
      [--kind={alpha|beta|gamma}]
      [--label=<text="demo">]
      [--src=<path> [--dst=<path>]]
      [--default={yes|no}]
      --commit
      (-f | --force)
      (--fmt | --format)
      (-q | --quiet | -v | --verbose)
      [--tag <label>]...

  .args:
    description: >-
      Positional forms - required, optional, optional-with-default,
      and greedy repeat (must be last).
    args: '<one> [<two>] [<three="3">] [<rest>...]'

  .pick:
    description: required choice - `(alt1|alt2|alt3)` becomes positional name `choice`
    args: '(north|south|east|west)'

  .mode:
    description: >-
      Positional choice in angle brackets - `<action={…}>` and
      optional `[<speed={…}='default'>]`.
    args: '<action={read|write|delete}> [<speed={fast|slow}="fast">]'

  .tail:
    description: optional repeating positional `[<items>...]`
    args: '[<items>...]'

  .group:
    description: >-
      Non-leaf group - invoke `x exapp demo group` with no subcommand to see auto-help.
    .alpha:
      description: nested leaf α - shares options via YAML anchor (see `reflect`)
      opts: &demo_verbose
        - '[-V | --verbose]'
      args: '<msg>'
    .beta:
      description: nested leaf β - same shared options as `alpha`
      opts: *demo_verbose
      args: '<msg>'

  .reflect:
    description: dump parsed options/args via `x-opts` / `x-args` (and `x-run`)
    opts: *demo_verbose
    args: '[<label="state">]'

  .sparse:
    description: optional repeat with a default (see also `tail` for plain `[<items>...]`)
    args: '[<parts="*">...]'

  .usage-tip:
    description: >-
      One-line handler - forwards to x-usage (same idea as README create → x-usage create).

  .io:
    description: interactive builtins
    .read:
      description: read a line of text
    .confirm:
      description: confirm a yes/no question
    .select:
      description: select an option from a list
    .multiselect:
      description: select an option from a list

  .style:
    description: terminal styling and TUI control builtins
    .prt:
      description: styled print via x-prt (--style)
    .tui:
      description: alternate buffer demo via x-tui

.debug:
  description: debug the environment variables
  $: |
    echo "DEBUG = $DEBUG"
handlers/exapp.handlers-a.yml 
# Scripts merged via `import.$` (first file). Keys are dotted command paths -
# import files keep this flat shape; they're not rewritten to dot-prefixed keys.

"":
  |
    topic=$(x-arg topic)
    note=$(x-opt global)
    echo "exapp - topic=${topic:-overview}  global-note=${note:-∅}"
    echo "Try:  x exapp --help"
    echo "       x exapp demo --help"

demo.opts:
  |
    echo "dry-run=$(x-opt dry-run)"
    echo "out=$(x-opt out)"
    echo "count=$(x-opt count)"
    mapfile -t defs < <(x-opt define)
    printf 'define[%s]\n' "${defs[@]:-}"
    echo "kind=$(x-opt kind)"
    echo "label=$(x-opt label)"
    echo "src=$(x-opt src)"
    echo "dst=$(x-opt dst)"
    echo "commit=$(x-opt commit)"
    echo "force=$(x-opt force)"
    echo "fmt=$(x-opt fmt)"
    echo "format=$(x-opt format)"
    echo "quiet=$(x-opt quiet)"
    echo "verbose=$(x-opt verbose)"
    mapfile -t tags < <(x-opt tag)
    printf 'tag[%s]\n' "${tags[@]:-}"

demo.mode:
  |
    echo "action=$(x-arg action)"
    echo "speed=$(x-arg speed)"

demo.args:
  |
    echo "one=$(x-arg one)"
    echo "two=$(x-arg two)"
    echo "three=$(x-arg three)"
    mapfile -t rest < <(x-arg rest)
    printf 'rest[%s]\n' "${rest[@]:-}"

demo.pick:
  |
    echo "direction=$(x-arg choice)"

demo.tail:
  |
    mapfile -t items < <(x-arg items)
    if ((${#items[@]})); then
      printf 'items: %s\n' "${items[@]}"
    else
      echo "(no items)"
    fi

demo.sparse:
  |
    mapfile -t parts < <(x-arg parts)
    echo "parts=${parts[*]}"

demo.usage-tip: x-usage demo.opts
handlers/exapp.handlers-b.yml 
# Second import file - merged keys must not duplicate the first import.

demo.group.alpha:
  |
    msg=$(x-arg msg)
    if [[ $(x-opt verbose) == true ]]; then
      echo "[verbose] alpha ← $msg"
    else
      echo "alpha ← $msg"
    fi

demo.group.beta:
  |
    msg=$(x-arg msg)
    if [[ $(x-opt verbose) == true ]]; then
      echo "[verbose] beta ← $msg"
    else
      echo "beta ← $msg"
    fi

demo.reflect:
  |
    declare -A opts args
    x-opts opts
    x-args args
    echo "--- x-opts ($(x-arg label)) ---"
    for k in "${!opts[@]}"; do echo "  $k=${opts[$k]}"; done | sort
    echo "--- x-args ---"
    for k in "${!args[@]}"; do echo "  $k=${args[$k]}"; done | sort
    echo "--- x-run ---"
    x-run echo "ran nested echo via x-run"

# IO commands
demo.io.read: |
  answer=$(x-io-read "What is your favorite color?")
  if [ "$answer" != "blue" ]; then
    echo "Wrong answer"
    exit 1
  else
    echo "Correct!"
  fi

demo.io.confirm: |
  answer=$(x-io-confirm "Are you sure you want to submit?" --default no)
  echo "Submit = $answer"
  answer=$(x-io-confirm "Are you sure you want to delete?" --default yes)
  echo "Delete = $answer"

demo.io.select: |
  x-io-select "Choose a color" \
    "red=Red" \
    "green=Green" \
    "blue=Blue" \
    "yellow/orange=Yellow or Orange" \
    -v color
  echo "Color = $color"

demo.io.multiselect: |
  x-io-select --multi "Choose a color" \
    "red=Red" \
    "green=Green" \
    "blue=Blue" \
    "yellow/orange=Yellow or Orange" \
    -v colors

  echo "First Color = ${colors[0]}"
  echo "All Colors = ${colors[*]}"

demo.style.prt: |
  x-prt \
    --style red,underline,bg-white \
    "Hello" \
    --style blue,bg-yellow \
    " World" \
    --newline
  x-prt --style green,bold "exapp prt demo"
  x-prt --style red,bold " exapp prt demo (red)" --newline

demo.style.tui: |
  x-tui --init --clear
  x-tui --home
  x-prt --style green,bold "exapp TUI demo"
  x-io-select "Action" "stay=Stay" "go=Go" -v action
  echo "Action = $action"
  if [ "$action" = "go" ]; then
    echo "Going!"
    x-prt --style red,bold "Going!"
  else
    echo "Staying!"
    x-prt --style green,bold "Staying!"
    sleep 2
    x-run x exapp demo style tui
  fi
  sleep 2
  x-tui --exit

Environment example (simple)

Demonstrates inline env: groups plus import.env/import.sh:

simple.x.yml 
name: simple
version: 0.0.0
description: an x app
import:
  $:
    - handlers/test.yml
  sh: handlers/example.sh
  env: .env.example

env:
  .env1:
    - MY_NAME: "env1"
  HELLO: "global"

.get-env:
  description: print the environment variable
  $: |
    echo "HELLO = $HELLO"
    x-env-load .env1
    echo "MY_NAME = $MY_NAME"
    echo "IS_DEBUG = $IS_DEBUG"

.test:
  description: test the environment variable

.hello-world:
  description: say hello to the world
  $: |
    hello_world

Typical workflows

  • Dotfiles scripts: x -i for personal utilities, x --ln for PATH access.
  • Monorepo tasks: root x.yml with .build, .test, .lint commands; no global install needed.
  • Team CLI tool: commit <tool>.x.yml to the repo; teammates run x tool subcmd from any subdirectory.
  • Complex internal CLI: global app in ~/.x.sh/apps/ with import.$ for large script sets.
